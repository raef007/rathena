// Auto-Battle Config Menu + Cash Shop
// Called when player uses Auto-Battle Capsule item (ID 30000)
// Also provides a cash shop NPC in Prontera
//
// NOTE: Item scripts cannot use interactive commands (next, select, menu, input)
// because pc_useitem() destroys the script state after run_script returns.
// The item uses doevent to hand off to this invisible NPC which has proper
// script state management.

// ========================================================
// Invisible NPC: Auto-Battle Config (handles item doevent)
// ========================================================
-	script	AutoBattleConfig	-1,{
	end;

OnUseItem:
	mes "[Auto-Battle Capsule]";
	mes "Configure your Auto-Battle settings.";
	next;

L_MainMenu:
	.@on = isautoattackon();
	.@toggle$ = (.@on) ? "Disable Auto-Attack" : "Enable Auto-Attack";
	.@menu = select(.@toggle$, "Configure Auto-Attack", "Close");
	if (.@menu == 1) {
		if (.@on) {
			atcommand "@autobattleoff";
			mes "[Auto-Battle]";
			mes "Auto-Attack: ^FF0000DISABLED^000000";
		} else {
			atcommand "@autoattack on";
			mes "[Auto-Battle]";
			mes "Auto-Attack: ^00FF00ENABLED^000000";
		}
		next;
		goto L_MainMenu;
	}
	if (.@menu == 3) close;

L_ConfigAttack:
	mes "[Configure Auto-Attack]";
	mes "Select a setting to adjust:";
	next;
	.@cfg = select("View Current Configuration", "Offensive Skill", "Target Filter", "Loot", "Regeneration", "Support Buffs", "Teleport", "Time Info", "Back");
	if (.@cfg == 1) goto L_Cfg_View;
	if (.@cfg == 2) goto L_Cfg_Skill;
	if (.@cfg == 3) goto L_Cfg_Target;
	if (.@cfg == 4) goto L_Cfg_Loot;
	if (.@cfg == 5) goto L_Cfg_Regen;
	if (.@cfg == 6) goto L_Cfg_SelfBuff;
	if (.@cfg == 7) goto L_Cfg_Teleport;
	if (.@cfg == 8) goto L_Cfg_Time;
	mes "[Auto-Battle Capsule]";
	mes "Configure your Auto-Battle settings.";
	next;
	goto L_MainMenu;

L_Cfg_View:
	mes "[Current Configuration]";
	mes "Full settings dumped to chat below.";
	mes "Scroll up to review.";
	next;
	atcommand "@autoattack";       // mode flags + range + pot + sit + target + skill + time
	atcommand "@autoattack target"; // detailed target whitelist
	atcommand "@autoattack skill";  // current offensive skill
	atcommand "@autosupport";      // support mode + skill/item counts
	atcommand "@autosupport list"; // detailed support skills + item buffs
	mes " ";
	mes "Anything missing? Re-configure from the menu.";
	next;
	goto L_ConfigAttack;
L_Cfg_Loot:
	mes "[Loot Settings]";
	mes "Toggle auto-loot pickup.";
	next;
	.@lsub = select("Enable Auto-Loot", "Disable Auto-Loot", "Back");
	if (.@lsub == 3) goto L_ConfigAttack;
	if (.@lsub == 1) {
		atcommand "@autoattack loot";
		mes "[Auto-Battle]";
		mes "Auto-Loot: ^00FF00ENABLED^000000";
	}
	if (.@lsub == 2) {
		atcommand "@autoattack loot off";
		mes "[Auto-Battle]";
		mes "Auto-Loot: ^FF0000DISABLED^000000";
	}
	next;
	goto L_ConfigAttack;

L_Cfg_Regen:
	mes "[Regeneration]";
	mes "Configure potions and sitting.";
	next;
	.@rsub = select("HP Recovery Item", "SP Recovery Item", "Sitting", "Go-Home", "Disable Potions", "Back");
	if (.@rsub == 6) goto L_ConfigAttack;
	if (.@rsub == 1) goto L_Regen_HPItem;
	if (.@rsub == 2) goto L_Regen_SPItem;
	if (.@rsub == 3) goto L_Cfg_Rest;
	if (.@rsub == 4) {
		mes "[Go-Home]";
		mes "Warp to save point when out of potions?";
		next;
		.@gh = select("Enable Go-Home", "Disable Go-Home");
		if (.@gh == 1)
			atcommand "@autoattack gohome";
		else
			atcommand "@autoattack gohome off";
		mes "[Auto-Battle]";
		mes "Go-Home: " + ((.@gh == 1) ? "^00FF00ENABLED^000000" : "^FF0000DISABLED^000000");
		next;
		goto L_Cfg_Regen;
	}
	if (.@rsub == 5) {
		atcommand "@autoattack pot off";
		mes "[Auto-Battle]";
		mes "Potions: ^FF0000DISABLED^000000";
		next;
		goto L_Cfg_Regen;
	}

L_Regen_HPItem:
	getinventorylist;
	.@hp_count = 0;
	for (.@i = 0; .@i < @inventorylist_count && .@hp_count < 80; .@i++) {
		.@id = @inventorylist_id[.@i];
		if (getiteminfo(.@id, ITEMINFO_TYPE) != IT_HEALING) continue;
		.@n$ = getitemname(.@id);
		// Exclude SP-only potions/herbs and the green status-cure family.
		if (compare(.@n$, "Green")) continue;
		if (compare(.@n$, "Blue Potion") || compare(.@n$, "Blue Herb")) continue;
		.@hp_ids[.@hp_count] = .@id;
		.@hp_names$[.@hp_count] = .@n$ + " x" + @inventorylist_amount[.@i];
		.@hp_count++;
	}
	if (.@hp_count == 0) {
		mes "[HP Recovery Item]";
		mes "No HP recovery items found in inventory.";
		next;
		goto L_Cfg_Regen;
	}
	.@m$ = "";
	for (.@i = 0; .@i < .@hp_count; .@i++) {
		if (.@i) .@m$ = .@m$ + ":";
		.@m$ = .@m$ + .@hp_names$[.@i];
	}
	.@m$ = .@m$ + ":Back";
	mes "[HP Recovery Item]";
	mes "Select one item.";
	next;
	.@pick = select(.@m$);
	if (.@pick == .@hp_count + 1) goto L_Cfg_Regen;
	.@chosen = .@hp_ids[.@pick - 1];
	atcommand "@autoattack pot " + .@chosen + " 90";
	mes "[Auto-Battle]";
	mes "HP recovery: ^0000FF" + getitemname(.@chosen) + "^000000 @ 90%";
	next;
	goto L_Cfg_Regen;

L_Regen_SPItem:
	getinventorylist;
	.@sp_count = 0;
	for (.@i = 0; .@i < @inventorylist_count && .@sp_count < 80; .@i++) {
		.@id = @inventorylist_id[.@i];
		if (getiteminfo(.@id, ITEMINFO_TYPE) != IT_HEALING) continue;
		.@n$ = getitemname(.@id);
		// compare() is a case-insensitive substring check (stristr) — NOT glob.
		// Don't wrap patterns in asterisks; literal "*" never appears in item
		// names so the match always fails.
		if (!(compare(.@n$, "Blue") || compare(.@n$, "Grape") || compare(.@n$, "Honey") ||
			compare(.@n$, "Yggdrasil") || compare(.@n$, "Strawberry") || compare(.@n$, "Lemon") ||
			compare(.@n$, "Seed") || compare(.@n$, "Berry") || compare(.@n$, "SP"))) continue;
		if (compare(.@n$, "Green")) continue;
		.@sp_ids[.@sp_count] = .@id;
		.@sp_names$[.@sp_count] = .@n$ + " x" + @inventorylist_amount[.@i];
		.@sp_count++;
	}
	if (.@sp_count == 0) {
		mes "[SP Recovery Item]";
		mes "No SP recovery items found in inventory.";
		next;
		goto L_Cfg_Regen;
	}
	.@m$ = "";
	for (.@i = 0; .@i < .@sp_count; .@i++) {
		if (.@i) .@m$ = .@m$ + ":";
		.@m$ = .@m$ + .@sp_names$[.@i];
	}
	.@m$ = .@m$ + ":Back";
	mes "[SP Recovery Item]";
	mes "Select one item.";
	next;
	.@pick = select(.@m$);
	if (.@pick == .@sp_count + 1) goto L_Cfg_Regen;
	.@chosen = .@sp_ids[.@pick - 1];
	atcommand "@autoattack sppot " + .@chosen + " 90";
	mes "[Auto-Battle]";
	mes "SP recovery: ^0000FF" + getitemname(.@chosen) + "^000000 @ 90%";
	next;
	goto L_Cfg_Regen;

L_Cfg_Teleport:
	mes "[Teleport]";
	mes "After ^00AA007 seconds without combat^000000 the bot can teleport to find new mobs.";
	mes " ";
	mes " ^0099FFFly Wing^000000: consumes one Fly Wing per teleport.";
	mes " ^0099FFTeleport Skill^000000: casts AL_TELEPORT (10 SP per cast).";
	mes "   - Works if you learned it OR a costume grants it.";
	mes " ";
	mes "Both can be on at once — Skill is tried first, Fly Wing is the fallback.";
	next;
	.@fw_on = (isautobattlemode(0x40) ? 1 : 0); // AUTOBATTLE_FLYWING
	.@tk_on = (isautobattlemode(0x100) ? 1 : 0); // AUTOBATTLE_TELESKILL
	.@fw$ = (.@fw_on) ? "Disable Fly Wing (currently ^00AA00ON^000000)" : "Enable Fly Wing (currently ^FF0000OFF^000000)";
	.@tk$ = (.@tk_on) ? "Disable Teleport Skill (currently ^00AA00ON^000000)" : "Enable Teleport Skill (currently ^FF0000OFF^000000)";
	.@tsub = select(.@fw$, .@tk$, "Back");
	if (.@tsub == 3) goto L_ConfigAttack;
	if (.@tsub == 1) atcommand (.@fw_on) ? "@autoattack flywing off" : "@autoattack flywing";
	if (.@tsub == 2) atcommand (.@tk_on) ? "@autoattack teleskill off" : "@autoattack teleskill";
	mes "[Auto-Battle]";
	mes "Teleport setting updated.";
	next;
	goto L_Cfg_Teleport;

L_Cfg_SelfBuff:
	mes "[Support Buffs]";
	mes "Configure self buffs from skills and items.";
	next;
	.@bmain = select("Skill Buffs", "Item Buffs", "View Status", "Back");
	if (.@bmain == 4) goto L_ConfigAttack;
	if (.@bmain == 3) {
		atcommand "@autosupport list";
		mes "[Support Buffs]";
		mes "Current buffs shown in chat.";
		next;
		goto L_Cfg_SelfBuff;
	}
	if (.@bmain == 1) goto L_Self_SkillBuffs;
	goto L_Self_ItemBuffs;

L_Self_SkillBuffs:
	getskilllist;
	.@buf_count = 0;
	// Cap of 80 — high-tier / all-skill characters can have 40+ buffs once
	// inherited Acolyte/Priest/Wizard/Knight buffs all match the filter.
	// Lower caps were silently truncating Monk skills like MO_CALLSPIRITS
	// because earlier buffs in the list filled all 30 slots first.
	for (.@i = 0; .@i < @skilllist_count && .@buf_count < 80; .@i++) {
		.@sid = @skilllist_id[.@i];
		if (!isautobattlebuffskill(.@sid)) continue;
		.@buf_ids[.@buf_count] = .@sid;
		.@buf_lvs[.@buf_count] = @skilllist_lv[.@i];
		.@buf_count++;
	}
	if (.@buf_count == 0) {
		mes "[Skill Buffs]";
		mes "No self buff skills found.";
		next;
		goto L_Cfg_SelfBuff;
	}
	for (.@i = 0; .@i < .@buf_count; .@i++) .@bsel[.@i] = 0;
L_SelfSkillToggle:
	.@m$ = "";
	for (.@i = 0; .@i < .@buf_count; .@i++) {
		if (.@i) .@m$ = .@m$ + ":";
		.@m$ = .@m$ + ((.@bsel[.@i]) ? "[X] " : "[ ] ") + getskillname(.@buf_ids[.@i]) + " Lv" + .@buf_lvs[.@i];
	}
	.@m$ = .@m$ + ":--- Apply Skill Buffs ---:--- Cancel ---";
	mes "[Skill Buffs]";
	mes "Toggle self buffs.";
	next;
	.@pick = select(.@m$);
	if (.@pick == .@buf_count + 2) goto L_Cfg_SelfBuff;
	if (.@pick == .@buf_count + 1) {
		atcommand "@autosupport clearbuffs";
		.@added = 0;
		for (.@i = 0; .@i < .@buf_count; .@i++) {
			if (.@bsel[.@i]) {
				atcommand "@autosupport add " + .@buf_ids[.@i] + " 100 0 1";
				.@added++;
			}
		}
		atcommand "@autosupport on";
		mes "[Support Buffs]";
		mes "^0000FF" + .@added + "^000000 skill buff(s) configured.";
		next;
		goto L_Cfg_SelfBuff;
	}
	.@idx = .@pick - 1;
	if (.@idx >= 0 && .@idx < .@buf_count) .@bsel[.@idx] = !.@bsel[.@idx];
	goto L_SelfSkillToggle;

L_Self_ItemBuffs:
	getinventorylist;
	.@item_count = 0;
	for (.@i = 0; .@i < @inventorylist_count && .@item_count < 40; .@i++) {
		.@id = @inventorylist_id[.@i];
		.@sc = autobattleitembuffstatus(.@id);
		if (.@sc <= 0) continue;
		.@item_ids[.@item_count] = .@id;
		.@item_sc[.@item_count] = .@sc;
		.@item_names$[.@item_count] = getitemname(.@id) + " x" + @inventorylist_amount[.@i];
		.@item_count++;
	}
	if (.@item_count == 0) {
		mes "[Item Buffs]";
		mes "No supported buff consumables found.";
		next;
		goto L_Cfg_SelfBuff;
	}
	for (.@i = 0; .@i < .@item_count; .@i++) .@isel[.@i] = 0;
L_SelfItemToggle:
	.@m$ = "";
	for (.@i = 0; .@i < .@item_count; .@i++) {
		if (.@i) .@m$ = .@m$ + ":";
		.@m$ = .@m$ + ((.@isel[.@i]) ? "[X] " : "[ ] ") + .@item_names$[.@i];
	}
	.@m$ = .@m$ + ":--- Apply Item Buffs ---:--- Cancel ---";
	mes "[Item Buffs]";
	mes "Toggle consumable buffs.";
	next;
	.@pick = select(.@m$);
	if (.@pick == .@item_count + 2) goto L_Cfg_SelfBuff;
	if (.@pick == .@item_count + 1) {
		atcommand "@autosupport clearitems";
		.@added = 0;
		for (.@i = 0; .@i < .@item_count; .@i++) {
			if (.@isel[.@i]) {
				atcommand "@autosupport additem " + .@item_ids[.@i] + " " + .@item_sc[.@i];
				.@added++;
			}
		}
		atcommand "@autosupport on";
		mes "[Support Buffs]";
		mes "^0000FF" + .@added + "^000000 item buff(s) configured.";
		next;
		goto L_Cfg_SelfBuff;
	}
	.@idx = .@pick - 1;
	if (.@idx >= 0 && .@idx < .@item_count) .@isel[.@idx] = !.@isel[.@idx];
	goto L_SelfItemToggle;

L_Cfg_Rest:
	mes "[Rest Settings]";
	mes "Auto-sit to regen when HP/SP is low.";
	next;
	.@rstsub = select("Set HP Rest", "Set SP Rest", "Set Both", "Disable Auto-Sit", "Back");
	if (.@rstsub == 5) goto L_ConfigAttack;
	if (.@rstsub == 1) {
		mes "[HP Rest]";
		mes "Sit when HP% drops below (0=disabled):";
		next;
		input(.@hp, 0, 90);
		atcommand "@autoattack sit " + .@hp + " 0";
		mes "[Auto-Battle]";
		if (.@hp > 0)
			mes "Auto-Sit HP: ^00FF00" + .@hp + "%^000000 (stand at " + (.@hp + 20) + "%)";
		else
			mes "HP Rest: ^FF0000DISABLED^000000";
	}
	if (.@rstsub == 2) {
		mes "[SP Rest]";
		mes "Sit when SP% drops below (0=disabled):";
		next;
		input(.@sp, 0, 90);
		atcommand "@autoattack sit 0 " + .@sp;
		mes "[Auto-Battle]";
		if (.@sp > 0)
			mes "Auto-Sit SP: ^00FF00" + .@sp + "%^000000 (stand at " + (.@sp + 20) + "%)";
		else
			mes "SP Rest: ^FF0000DISABLED^000000";
	}
	if (.@rstsub == 3) {
		mes "[Rest Thresholds]";
		mes "Sit when HP% drops below (0=disabled):";
		next;
		input(.@hp, 0, 90);
		mes "Sit when SP% drops below (0=disabled):";
		next;
		input(.@sp, 0, 90);
		atcommand "@autoattack sit " + .@hp + " " + .@sp;
		mes "[Auto-Battle]";
		mes "Auto-Sit: HP<" + .@hp + "% SP<" + .@sp + "%";
	}
	if (.@rstsub == 4) {
		atcommand "@autoattack sit off";
		mes "[Auto-Battle]";
		mes "Auto-Sit: ^FF0000DISABLED^000000";
	}
	next;
	goto L_ConfigAttack;

L_Cfg_Target:
	mes "[Target Filter]";
	mes "Choose which monsters to attack.";
	next;
	.@tsub = select("Target All Monsters", "Select Specific Monsters", "View Current Targets", "Back");
	if (.@tsub == 4) goto L_ConfigAttack;
	if (.@tsub == 1) {
		atcommand "@autoattack target all";
		mes "[Auto-Battle]";
		mes "Target: ^00FF00All monsters^000000";
		next;
		goto L_ConfigAttack;
	}
	if (.@tsub == 3) {
		atcommand "@autoattack target";
		mes "[Auto-Battle]";
		mes "Current targets shown in chat.";
		next;
		goto L_ConfigAttack;
	}
	.@mapname$ = strcharinfo(3);
	.@mob_count = getmapunits(BL_MOB, .@mapname$, .@mob_gids[0]);
	.@unique_count = 0;
	freeloop(1);
	for (.@i = 0; .@i < .@mob_count && .@i < 200; .@i++) {
		if (.@mob_gids[.@i] <= 0) continue;
		if (!unitexists(.@mob_gids[.@i])) continue;
		getunitdata .@mob_gids[.@i], .@unitdata;
		.@class = .@unitdata[UMOB_CLASS];
		if (.@class <= 0) continue;
		if (.@unitdata[UMOB_AI] > 0) continue;
		.@dupe = 0;
		for (.@j = 0; .@j < .@unique_count; .@j++) {
			if (.@unique_ids[.@j] == .@class) {
				.@dupe = 1;
				break;
			}
		}
		if (!.@dupe && .@unique_count < 20) {
			.@unique_ids[.@unique_count] = .@class;
			.@name$ = getmonsterinfo(.@class, MOB_NAME);
			if (.@name$ == "" || .@name$ == "null") .@name$ = "Mob#" + .@class;
			.@unique_names$[.@unique_count] = .@name$;
			.@unique_count++;
		}
	}
	freeloop(0);
	if (.@unique_count == 0) {
		mes "[Target Filter]";
		mes "No monsters found on this map.";
		next;
		goto L_ConfigAttack;
	}
	for (.@i = 0; .@i < .@unique_count; .@i++)
		.@selected[.@i] = 0;

L_TargetToggle:
	.@tmenu$ = "";
	for (.@i = 0; .@i < .@unique_count; .@i++) {
		if (.@i > 0) .@tmenu$ = .@tmenu$ + ":";
		if (.@selected[.@i])
			.@tmenu$ = .@tmenu$ + "[X] " + .@unique_names$[.@i] + " (" + .@unique_ids[.@i] + ")";
		else
			.@tmenu$ = .@tmenu$ + "[ ] " + .@unique_names$[.@i] + " (" + .@unique_ids[.@i] + ")";
	}
	.@tmenu$ = .@tmenu$ + ":--- Apply Selection ---:--- Cancel ---";
	mes "[Target Filter]";
	mes "Toggle monsters to target:";
	mes "^0000FF[X]^000000 = selected  ^999999[ ]^000000 = skip";
	next;
	.@pick = select(.@tmenu$);
	if (.@pick == .@unique_count + 2) {
		mes "[Auto-Battle]";
		mes "Target filter unchanged.";
		next;
		goto L_ConfigAttack;
	}
	if (.@pick == .@unique_count + 1) {
		atcommand "@autoattack target clear";
		.@applied = 0;
		for (.@i = 0; .@i < .@unique_count; .@i++) {
			if (.@selected[.@i]) {
				atcommand "@autoattack target " + .@unique_ids[.@i];
				.@applied++;
			}
		}
		mes "[Auto-Battle]";
		if (.@applied == 0)
			mes "Target: ^00FF00All monsters^000000 (none selected)";
		else
			mes "Target: ^0000FF" + .@applied + " specific mob type(s)^000000";
		next;
		goto L_ConfigAttack;
	}
	.@idx = .@pick - 1;
	if (.@idx >= 0 && .@idx < .@unique_count)
		.@selected[.@idx] = !.@selected[.@idx];
	goto L_TargetToggle;

L_Cfg_Skill:
	mes "[Skill Settings]";
	mes "Choose your combat skill.";
	next;
	.@ssub = select("Normal Attack Only", "Select Offensive Skill", "View Current Skill", "Back");
	if (.@ssub == 4) goto L_ConfigAttack;
	if (.@ssub == 1) {
		atcommand "@autoattack skill off";
		mes "[Auto-Battle]";
		mes "Attack: ^00FF00Normal Attack^000000";
		next;
		goto L_ConfigAttack;
	}
	if (.@ssub == 3) {
		atcommand "@autoattack skill";
		mes "[Auto-Battle]";
		mes "Current skill shown in chat.";
		next;
		goto L_ConfigAttack;
	}
	getskilllist;
	.@off_count = 0;
	for (.@i = 0; .@i < @skilllist_count; .@i++) {
		.@sid = @skilllist_id[.@i];
		.@slv = @skilllist_lv[.@i];
		if (!isautobattleattackskill(.@sid)) continue;
		.@off_ids[.@off_count] = .@sid;
		.@off_lvs[.@off_count] = .@slv;
		.@off_count++;
		if (.@off_count >= 30) break;
	}
	if (.@off_count == 0) {
		mes "[Skill Settings]";
		mes "No offensive skills found.";
		next;
		goto L_ConfigAttack;
	}
	.@smenu$ = "Normal Attack Only";
	for (.@i = 0; .@i < .@off_count; .@i++)
		.@smenu$ = .@smenu$ + ":" + getskillname(.@off_ids[.@i]) + " Lv" + .@off_lvs[.@i];
	.@smenu$ = .@smenu$ + ":Back";
	mes "[Select Skill]";
	mes "Pick your offensive skill:";
	next;
	.@spick = select(.@smenu$);
	if (.@spick == 1) {
		atcommand "@autoattack skill off";
		mes "[Auto-Battle]";
		mes "Attack: ^00FF00Normal Attack^000000";
	} else if (.@spick <= .@off_count + 1) {
		.@chosen_id = .@off_ids[.@spick - 2];
		.@chosen_lv = .@off_lvs[.@spick - 2];
		atcommand "@autoattack skill " + .@chosen_id + " " + .@chosen_lv;
		mes "[Auto-Battle]";
		mes "Skill: ^0000FF" + getskillname(.@chosen_id) + " Lv" + .@chosen_lv + "^000000";
	}
	next;
	goto L_ConfigAttack;

L_Cfg_Time:
	atcommand "@autoattack time";
	mes "[Auto-Battle]";
	mes "Time info displayed in chat window.";
	next;
	goto L_ConfigAttack;

	end;
}

// ========================================================
// Function: F_AutoBattleMenu (called from item script)
// ========================================================
function	script	F_AutoBattleMenu	{
	doevent "AutoBattleConfig::OnUseItem";
	end;
}

// ========================================================
// Invisible NPC: Auto-Support Config (Phase 25)
// ========================================================
-	script	AutoSupportConfig	-1,{
	end;

OnUseItem:
	mes "[Auto-Support Capsule]";
	mes "Configure party support automation.";
	next;

L_SupportMain:
	.@menu = select("Enable Auto-Support", "Configure Auto-Support", "Stop All", "Close");
	if (.@menu == 1) {
		atcommand "@autosupport on";
		mes "[Auto-Support]";
		mes "Auto-Support: ^00FF00ENABLED^000000";
		next;
		goto L_SupportMain;
	}
	if (.@menu == 3) {
		atcommand "@autosupport off";
		mes "[Auto-Support]";
		mes "Auto-Support: ^FF0000STOPPED^000000";
		close;
	}
	if (.@menu == 4) close;

L_ConfigSupport:
	mes "[Configure Auto-Support]";
	mes "Select a setting to adjust:";
	next;
	.@scfg = select("Heal Settings", "Buff Settings", "Target Settings", "View Status", "Back");
	if (.@scfg == 1) goto L_Sup_Heal;
	if (.@scfg == 2) goto L_Sup_Buff;
	if (.@scfg == 3) goto L_Sup_Target;
	if (.@scfg == 4) goto L_Sup_Status;
	mes "[Auto-Support Capsule]";
	mes "Configure party support automation.";
	next;
	goto L_SupportMain;
L_Sup_Heal:
	.@hlv = getskilllv(28);
	if (.@hlv == 0) {
		mes "[Heal Settings]";
		mes "Your character does not have the ^FF0000Heal^000000 skill.";
		next;
		goto L_ConfigSupport;
	}
	mes "[Heal Settings]";
	mes "Auto-cast ^0000FFHeal Lv" + .@hlv + "^000000 on injured targets.";
	mes "Set the HP threshold below which Heal triggers.";
	next;
	mes "[HP Threshold]";
	mes "Cast Heal when target HP drops below (%):";
	next;
	input(.@hthresh, 1, 100);
	mes "[Target Scope]";
	mes "Who to heal?";
	next;
	.@hscope = select("Self only", "Party members") - 1;
	atcommand "@autosupport add 28 " + .@hthresh + " " + .@hscope;
	mes "[Auto-Support]";
	mes "^0000FFHeal Lv" + .@hlv + "^000000 configured.";
	mes "Trigger: HP < " + .@hthresh + "%";
	mes "Scope: " + ((.@hscope == 0) ? "Self" : "Party");
	next;
	goto L_ConfigSupport;

L_Sup_Buff:
	mes "[Buff Settings]";
	mes "Select buff skills to auto-cast.";
	next;
	getskilllist;
	.@buf_count = 0;
	for (.@i = 0; .@i < @skilllist_count; .@i++) {
		.@sid = @skilllist_id[.@i];
		.@slv = @skilllist_lv[.@i];
		if (isautobattlebuffskill(.@sid)) {
			.@buf_ids[.@buf_count] = .@sid;
			.@buf_lvs[.@buf_count] = .@slv;
			.@buf_count++;
			// Same reasoning as L_Self_SkillBuffs — high-tier/all-skill
			// characters easily exceed 20 buffs.
			if (.@buf_count >= 80) break;
		}
	}
	if (.@buf_count == 0) {
		mes "No buff skills found.";
		next;
		goto L_ConfigSupport;
	}
	for (.@i = 0; .@i < .@buf_count; .@i++)
		.@bsel[.@i] = 0;

L_BuffToggle:
	.@bmenu$ = "";
	for (.@i = 0; .@i < .@buf_count; .@i++) {
		if (.@i > 0) .@bmenu$ = .@bmenu$ + ":";
		if (.@bsel[.@i])
			.@bmenu$ = .@bmenu$ + "[X] " + getskillname(.@buf_ids[.@i]) + " Lv" + .@buf_lvs[.@i];
		else
			.@bmenu$ = .@bmenu$ + "[ ] " + getskillname(.@buf_ids[.@i]) + " Lv" + .@buf_lvs[.@i];
	}
	.@bmenu$ = .@bmenu$ + ":--- Apply Buffs ---:--- Cancel ---";
	mes "[Buff Settings]";
	mes "Toggle buffs to auto-cast:";
	mes "^0000FF[X]^000000 = active  ^999999[ ]^000000 = inactive";
	next;
	.@bpick = select(.@bmenu$);
	if (.@bpick == .@buf_count + 2) {
		mes "Cancelled.";
		next;
		goto L_ConfigSupport;
	}
	if (.@bpick == .@buf_count + 1) {
		atcommand "@autosupport clearbuffs";
		.@added = 0;
		for (.@i = 0; .@i < .@buf_count; .@i++) {
			if (.@bsel[.@i]) {
				atcommand "@autosupport add " + .@buf_ids[.@i] + " 100 1 1";
				.@added++;
			}
		}
		mes "[Auto-Support]";
		mes "^0000FF" + .@added + "^000000 buff skill(s) configured.";
		next;
		goto L_ConfigSupport;
	}
	.@bidx = .@bpick - 1;
	if (.@bidx >= 0 && .@bidx < .@buf_count)
		.@bsel[.@bidx] = !.@bsel[.@bidx];
	goto L_BuffToggle;

L_Sup_Target:
	mes "[Target Settings]";
	mes "Who should you support?";
	next;
	.@tsub = select("Auto-lock Nearby Member", "Party Leader Only", "Specific Member", "Back");
	if (.@tsub == 4) goto L_ConfigSupport;
	if (.@tsub == 1) {
		atcommand "@autosupport target all";
		mes "[Auto-Support]";
		mes "Target: ^00FF00Auto-lock nearby member^000000";
	}
	if (.@tsub == 2) {
		atcommand "@autosupport target leader";
		mes "[Auto-Support]";
		mes "Target: ^00FF00Party leader only^000000";
	}
	if (.@tsub == 3) {
		mes "Enter player name:";
		next;
		input(.@pname$);
		atcommand "@autosupport target " + .@pname$;
		mes "[Auto-Support]";
		mes "Target: ^0000FF" + .@pname$ + "^000000";
	}
	next;
	goto L_ConfigSupport;

L_Sup_Status:
	atcommand "@autosupport";
	atcommand "@autosupport list";
	atcommand "@autosupport target";
	mes "[Auto-Support]";
	mes "Status shown in chat window.";
	next;
	goto L_ConfigSupport;

	end;
}
