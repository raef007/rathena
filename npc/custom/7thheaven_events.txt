// ==========================================================================
// 7th Heaven — Persistent Leash & Passive Event Handler
// ==========================================================================
// Applies Beast Master's Leash bonuses via bonus_script — works regardless of
// other SCs the player may have, no skill-level math, no mount requirement.
// Re-applied on login and on every leash use.
// ==========================================================================

-	script	7thHeaven_Events	-1,{

OnPCLoginEvent:
	// ---- Innkeeper Mara's tavern blessings (1 hour, persistent) ----
	// INN_BUFF_END is absolute server tick — the clock ticks down even while
	// the player is offline. On login, re-apply with whatever time remains.
	if (INN_BUFF_END > 0) {
		.@remain = INN_BUFF_END - gettimetick(2);
		if (.@remain > 0) {
			.@remain_ms = .@remain * 1000;
			if (getstatus(SC_MAGNIFICAT) == 0)
				sc_start SC_MAGNIFICAT, .@remain_ms, 5;
			if (getstatus(SC_INCAGI) == 0)
				sc_start SC_INCAGI, .@remain_ms, 10;
			if (getstatus(SC_GLORIA) == 0)
				sc_start SC_GLORIA, .@remain_ms, 5;
		} else {
			set INN_BUFF_END, 0;
		}
	}

	doevent "7thHeaven_Events::OnApplyLeash";
	end;

// Apply leash bonuses based on current LEASH_LV.
// Called from OnPCLoginEvent and from the leash item's Script.
//
// Per level, applies (always — leash in inventory is sufficient):
//   +N% MaxHP / MaxSP, +N% HP/SP regen, +200*Lv weight limit
// Lv9 adds +1 to all stats, Lv10 adds +2.
//
// Uses bonus_script flag 1024 (replace duplicates by extending duration) so
// re-applying refreshes cleanly without stacking. 23h duration means the
// bonus survives a normal play session; login + item-use keep it topped up.
OnApplyLeash:
	if (countitem(50011) <= 0) end;
	if (LEASH_LV <= 0) end;

	.@lv = LEASH_LV;
	if (.@lv > 10) .@lv = 10;

	// Per-level scaling tables — index 1..10 (index 0 unused).
	setarray .@hp[0],     0, 3, 5, 8, 11, 14, 17, 20, 23, 26, 30;
	setarray .@sp[0],     0, 3, 5, 8, 11, 14, 17, 20, 23, 26, 30;
	setarray .@hpregen[0],0, 5,10,15, 20, 25, 30, 35, 40, 45, 50;
	setarray .@spregen[0],0, 5,10,15, 20, 25, 30, 35, 40, 45, 50;

	.@maxhp_pct  = .@hp[.@lv];
	.@maxsp_pct  = .@sp[.@lv];
	.@hpreg_pct  = .@hpregen[.@lv];
	.@spreg_pct  = .@spregen[.@lv];
	.@weight     = 200 * .@lv;
	.@allstat    = (.@lv >= 9) ? (.@lv - 8) : 0;

	.@script$ = "{ "
		+ "bonus bMaxHPrate," + .@maxhp_pct + "; "
		+ "bonus bMaxSPrate," + .@maxsp_pct + "; "
		+ "bonus bHPrecovRate," + .@hpreg_pct + "; "
		+ "bonus bSPrecovRate," + .@spreg_pct + "; "
		+ "bonus bAddMaxWeight," + .@weight + "; ";
	if (.@allstat > 0)
		.@script$ = .@script$ + "bonus bAllStats," + .@allstat + "; ";
	.@script$ = .@script$ + "}";

	// 23h duration, flag 1024 = replace/extend duplicates, type 1 = buff.
	bonus_script .@script$, 82800, 1024, 1;
	end;
}
