// ==========================================================================
// 7th Heaven — Persistent Leash & Passive Event Handler
// ==========================================================================
// Re-applies Beast Master's Leash buffs on every login AND on mount-up.
// Lv1:    starter +5% HP while mounted
// Lv2-8:  scaling HP%, regen, def/mdef/flee while mounted
// Lv9-10: passive all-stat bonus (no mount required, leash in inventory)
// ==========================================================================

-	script	7thHeaven_Events	-1,{

OnPCLoginEvent:
	// ---- Innkeeper Mara's tavern blessings (1 hour, persistent) ----
	// INN_BUFF_END is absolute server tick — the clock ticks down even while
	// the player is offline. On login, re-apply with whatever time remains.
	if (INN_BUFF_END > 0) {
		.@remain = INN_BUFF_END - gettimetick(2);
		if (.@remain > 0) {
			.@remain_ms = .@remain * 1000;
			if (getstatus(SC_MAGNIFICAT) == 0)
				sc_start SC_MAGNIFICAT, .@remain_ms, 5;
			if (getstatus(SC_INCAGI) == 0)
				sc_start SC_INCAGI, .@remain_ms, 10;
			if (getstatus(SC_GLORIA) == 0)
				sc_start SC_GLORIA, .@remain_ms, 5;
		} else {
			set INN_BUFF_END, 0;
		}
	}

	doevent "7thHeaven_Events::OnApplyLeash";
	end;

// Apply leash buffs based on current LEASH_LV and mount state.
// Called from OnPCLoginEvent and from the leash item's Script (after setmounting).
OnApplyLeash:
	if (countitem(50011) <= 0) end;
	if (LEASH_LV <= 0) end;

	// ---- Passive (Lv9-10) — leash in inventory, no mount required ----
	if (LEASH_LV >= 9) {
		.@val = LEASH_LV - 8; // Lv9 = +1, Lv10 = +2
		sc_start SC_INCALLSTATUS, 82800000, .@val;
	}

	// ---- While-mounted buffs (Lv1-8) ----
	if (checkriding() || ismounting()) {
		.@lv = LEASH_LV;

		// HP bonus: Lv1 = +5%, Lv2-4 = +6/8/10%, Lv5+ capped at +12%
		setarray .@hp_table[0], 0, 5, 6, 8, 10, 12, 12, 12, 12;
		.@hp_pct = .@hp_table[(.@lv > 8) ? 8 : .@lv];
		if (.@hp_pct > 0)
			sc_start SC_APPLEIDUN, 82800000, .@hp_pct;

		// Regen boost (Magnificat proxy): scales with leash level, cap at 5
		.@regen_lv = (.@lv >= 5) ? 5 : .@lv;
		if (.@regen_lv > 0)
			sc_start SC_MAGNIFICAT, 82800000, .@regen_lv;

		// Def/MDef/Flee bonus (Lv6-8): +10/20/30
		if (.@lv >= 6) {
			.@stat_bonus = (.@lv - 5) * 10;
			sc_start SC_DEF_RATE, 82800000, .@stat_bonus;
			sc_start SC_MDEF_RATE, 82800000, .@stat_bonus;
			sc_start SC_INCFLEE, 82800000, .@stat_bonus;
		}
	}
	end;
}
