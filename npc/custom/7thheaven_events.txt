// ==========================================================================
// 7th Heaven — Persistent Leash & Passive Event Handler
// ==========================================================================
// Lv1-8 bonuses apply ONLY while mounted (HP regen, SP regen, weight).
// Lv9-10 bonuses apply whenever the leash is in inventory (+1/+2 all stats).
//
// Bonuses use bonus_script (not status changes) to avoid SC icon residue and
// MaxHP-not-recalculating bugs. F_ApplyLeash is called synchronously via
// callfunc from both OnPCLoginEvent and the leash item's Script.
// ==========================================================================

// --------------------------------------------------------------------------
// F_ApplyLeash — synchronous callfunc target.
//
// Steps:
//   1. Clear all existing bonus_scripts on the player.
//   2. If leash absent or LEASH_LV == 0, stop.
//   3. If LEASH_LV >= 9, apply +AllStats (always-on).
//   4. If mounted AND LEASH_LV in 1..8, apply HP/SP regen + weight.
//
// callfunc keeps the invoking player attached, unlike doevent which queues.
// --------------------------------------------------------------------------
function	script	F_ApplyLeash	{
	bonus_script_clear;

	if (countitem(50011) <= 0) return;
	if (LEASH_LV <= 0) return;

	.@lv = LEASH_LV;
	if (.@lv > 10) .@lv = 10;

	// ---- Lv9-10 always-on: +AllStats ----
	if (.@lv >= 9) {
		.@allstat = .@lv - 8; // 9->1, 10->2
		bonus_script "{ bonus bAllStats," + .@allstat + "; }", 82800, 1024, 1;
	}

	// ---- Lv1-8 mounted only: regen + weight ----
	if (.@lv > 8) return;
	if (!checkriding() && !ismounting()) return;

	.@regen  = .@lv * 5;   // Lv1=5%, Lv8=40%
	.@weight = .@lv * 200; // Lv1=200, Lv8=1600

	bonus_script "{ bonus bHPrecovRate," + .@regen + "; bonus bSPrecovRate," + .@regen + "; bonus bAddMaxWeight," + .@weight + "; }", 82800, 1024, 1;
	return;
}

// --------------------------------------------------------------------------
// 7thHeaven_Events — login hooks for Innkeeper buffs and leash refresh.
// --------------------------------------------------------------------------
-	script	7thHeaven_Events	-1,{

OnPCLoginEvent:
	// ---- Innkeeper Mara's tavern blessings (1 hour, persistent) ----
	if (INN_BUFF_END > 0) {
		.@remain = INN_BUFF_END - gettimetick(2);
		if (.@remain > 0) {
			.@remain_ms = .@remain * 1000;
			if (getstatus(SC_MAGNIFICAT) == 0)
				sc_start SC_MAGNIFICAT, .@remain_ms, 5;
			if (getstatus(SC_INCAGI) == 0)
				sc_start SC_INCAGI, .@remain_ms, 10;
			if (getstatus(SC_GLORIA) == 0)
				sc_start SC_GLORIA, .@remain_ms, 5;
		} else {
			set INN_BUFF_END, 0;
		}
	}

	callfunc "F_ApplyLeash";
	end;
}
