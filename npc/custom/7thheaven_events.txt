// ==========================================================================
// 7th Heaven — Persistent Leash & Passive Event Handler
// ==========================================================================
// Lv1-8 bonuses apply ONLY while mounted (HP%/SP%, regen, weight).
// Lv9-10 bonuses apply whenever the leash is in inventory (+1/+2 all stats).
//
// Both bonuses use bonus_script. F_ApplyLeash is called synchronously via
// callfunc from OnPCLoginEvent and from the leash item's Script.
//
// EFST_FOOD_STR is reused as a visible status icon so the player can confirm
// the bonus is active.
// ==========================================================================

// --------------------------------------------------------------------------
// F_ApplyLeash — synchronous callfunc target.
// --------------------------------------------------------------------------
function	script	F_ApplyLeash	{
	// Wipe prior leash bonuses so dismounting cleans up. (No other system on
	// this server uses bonus_script for players; if that changes, switch to
	// flag-tagged clears.)
	bonus_script_clear;

	if (countitem(50011) <= 0)
		return;
	if (LEASH_LV <= 0)
		return;

	.@lv = LEASH_LV;
	if (.@lv > 10) .@lv = 10;

	// ---- Lv9-10 always-on: +AllStats ----
	if (.@lv >= 9) {
		.@allstat = .@lv - 8;
		bonus_script "{ bonus bAllStats," + .@allstat + "; }", 82800, 1024, 1, EFST_FOOD_STR;
	}

	// ---- Lv1-8 mounted only ----
	if (.@lv > 8)
		return;

	if (!checkriding() && !ismounting())
		return;

	.@hp_pct  = .@lv * 2;   // Lv1=2%, Lv8=16%
	.@sp_pct  = .@lv * 2;
	.@regen   = .@lv * 5;   // Lv1=5%, Lv8=40%
	.@weight  = .@lv * 200; // Lv1=200, Lv8=1600

	bonus_script "{ bonus bMaxHPrate," + .@hp_pct + "; bonus bMaxSPrate," + .@sp_pct + "; bonus bHPrecovRate," + .@regen + "; bonus bSPrecovRate," + .@regen + "; bonus bAddMaxWeight," + .@weight + "; }", 82800, 1024, 1, EFST_FOOD_STR;
	return;
}

// --------------------------------------------------------------------------
// 7thHeaven_Events — login hooks for Innkeeper buffs and leash refresh.
// --------------------------------------------------------------------------
-	script	7thHeaven_Events	-1,{

OnPCLoginEvent:
	// ---- Innkeeper Mara's tavern blessings (1 hour, persistent) ----
	if (INN_BUFF_END > 0) {
		.@remain = INN_BUFF_END - gettimetick(2);
		if (.@remain > 0) {
			.@remain_ms = .@remain * 1000;
			if (getstatus(SC_MAGNIFICAT) == 0)
				sc_start SC_MAGNIFICAT, .@remain_ms, 5;
			if (getstatus(SC_INCAGI) == 0)
				sc_start SC_INCAGI, .@remain_ms, 10;
			if (getstatus(SC_GLORIA) == 0)
				sc_start SC_GLORIA, .@remain_ms, 5;
		} else {
			set INN_BUFF_END, 0;
		}
	}

	callfunc "F_ApplyLeash";
	end;
}
